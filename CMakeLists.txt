cmake_minimum_required(VERSION 3.24)
project(pokerstove VERSION 1.2 LANGUAGES CXX)

if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build/release")
endif()

option(BUILD_SHARED_LIBS "Build shared core libraries" OFF)
option(MODERN_POKERSTOVE_BUILD_APPS "Build command-line tools" ${PROJECT_IS_TOP_LEVEL})
option(MODERN_POKERSTOVE_BUILD_PYTHON "Build SWIG Python bindings" OFF)
option(BUILD_TESTING "Build PokerStove tests" ${PROJECT_IS_TOP_LEVEL})
include(CTest)
include(GNUInstallDirs)

include(cmake/dependencies.cmake)
include(cmake/libraries.cmake)
if(MODERN_POKERSTOVE_BUILD_APPS)
    include(cmake/applications.cmake)
endif()
if(BUILD_TESTING)
    include(cmake/tests.cmake)
endif()
if(MODERN_POKERSTOVE_BUILD_PYTHON)
    include(cmake/python.cmake)
endif()
include(cmake/install.cmake)
